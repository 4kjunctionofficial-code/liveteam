function runSingleTicker(elementId, textToShow) { const el = document.getElementById(elementId); if (!el) return; if ( el.dataset.currentText === textToShow && el.dataset.language === String(isEnglish) ) { return; } el.dataset.currentText = textToShow; el.dataset.language = String(isEnglish); el.className = "anim-wrap-single"; el.innerHTML = ""; el.style.removeProperty( "--marquee-duration" ); el.style.removeProperty( "--scroll-distance" ); const textNode = document.createElement("span"); textNode.className = "ticker-text"; textNode.innerHTML = textToShow; el.appendChild(textNode); requestAnimationFrame(() => { const boxWidth = el.clientWidth; const textWidth = textNode.scrollWidth; if (textWidth > boxWidth - 20) { const distance = textWidth - boxWidth + 30; const duration = Math.max( 12, distance / 75 + 2 ); el.classList.add( "marquee-mode" ); el.style.setProperty( "--scroll-distance", `${distance}px` ); el.style.setProperty( "--marquee-duration", `${duration}s` ); textNode.style.animation = "none"; void textNode.offsetWidth; textNode.style.animation = `tickerMarquee ${duration}s linear infinite`; } else { el.classList.add( "anim-center-in" ); } }); }
